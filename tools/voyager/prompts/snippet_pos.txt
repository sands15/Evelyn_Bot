235: async function waitForFeetPosition(timeoutMs = 250, fallback = null) {
236:   const deadline = Date.now() + Math.max(0, Number(timeoutMs || 0));
237:   while (Date.now() <= deadline) {
238:     const feet = getFeetPosition();
239:     if (feet) return feet;
240:     if (Date.now() >= deadline) break;
241:     await delay(50);

243:   return fallback;
